# 内部データスキーマの定義

googleもoutlookも、これに合わせてパースして処理する。

日時のタイムゾーンに関しては、YYYY-MM-DDThh:mm:ss形式で保存し、それとAsia/Tokyoなどのタイムゾーン情報と共に保存することで、時間を保証する。

```json
{
	"id": "abc123...", // このイベント自身のプラットフォーム上のID

	"subject": "毎週月曜10時のMTG",
	"description": "このイベントの概要欄",
	"location": "Zoom meeting",
	"isAllDay": true,
	"timeZone": "Asia/Tokyo", // このイベント（親）が使う唯一のタイムゾーン。exception側も基本これを継承
	"start": "2026-09-14T10:00:00", // ローカル時刻表記（timeZone基準）。isAllDay=trueの場合は時刻部分は00:00:00固定
	"end": "2026-09-14T11:00:00", // isAllDay=trueの場合は開始日の翌日0時（Google/Graphとも「終了は排他的」の仕様に合わせる）
	"showAs": "busy", // 取り込み中か空き時間かを表す
	"sensitivity": "public", // イベントが公開か非公開かを表す
	"updatedAt": "2026-09-01T12:00:00Z", // このイベント自身の最終更新日時（プラットフォームのupdated/lastModifiedDateTimeをそのまま格納）
	// 子（exception）の変更はこの値に影響しない想定（Google/Graphとも親子は別オブジェクトのため）。
	// ただし両API共に「子の変更で親のupdatedAtが変わらないこと」を明言した公式記述はないため、
	// 実装時に実際の環境で一度動作確認しておくこと。

	// イベントの繰り返しに関する情報
	"recurrence": {
		"pattern": {
			"type": "weekly", // daily / weekly / monthly / yearly
			"interval": 1, // 単位ごとの間隔。毎週=1、隔週=2、毎月なら1、隔月なら2、というように「何単位ごとか」を表す
			"daysOfWeek": ["monday"], // type=weekly、またはmonthly/yearlyの相対指定（weekIndex使用時）で使用。
			// daily、またはmonthly/yearlyの絶対日付指定（dayOfMonth使用）では空配列
			"weekIndex": "", // "first"/"second"/"third"/"fourth"/"last"のいずれか。
			// monthly/yearlyで「第n曜日」を指定する場合のみ使用。空文字なら未使用（dayOfMonthを使う）
			"dayOfMonth": null, // monthly/yearlyで「その月の何日」を絶対指定する場合に使用。weekIndexとは排他利用（両方同時に値を持たせない）
			"count": null, // 終了条件（回数指定）。指定時はendDateはnullにする
			"endDate": null // 終了条件（日付指定）。日付のみを持つ（時刻はstartの時刻をそのまま使う）。
			// countとendDateがともにnullなら無期限（noEnd）
			// ※count/endDateどちらか一方のみ設定可能。両方nullで無期限、片方だけ設定というルールで統一する
		},

		// --- 親の繰り返しパターン自体が変更された場合の扱い（重要） ---
		// Google Calendar API、Microsoft Graph APIともに、既存の繰り返しシリーズの
		// pattern（曜日・頻度・終了条件など）を直接書き換えた場合、既存のexceptionが
		// どう扱われるかを保証する公式記述が見当たらなかった。
		// そのため、このシステムでは「パターン変更」を検知した場合、
		// 親の直接書き換えは行わず、必ず以下の「シリーズ分割」方式で処理する。
		//
		//   1. 既存シリーズを、変更が適用され始める直前の時点で打ち切る
		//      （Google: RRULEにUNTIL/COUNTを設定してpatch、
		//       Graph: recurrence.range.typeをendDateに変更してPATCH）
		//   2. 変更後の新しいpatternで、新しいシリーズを新規作成する
		//   3. 打ち切った旧シリーズに属する既存exceptionはそのまま旧シリーズに残し、
		//      変更対象にしない
		//   4. 新シリーズは最初からexceptionが空の状態でスタートする
		//
		// これはGoogle公式が推奨する「シリーズ分割」の手順に両プラットフォームとも
		// 合わせることで、未保証の挙動を避けるための方針。
		//
		// 新シリーズの開始日（分割点）は、反映元側の「現在のpattern」から逆算して
		// 決定する（反映元がすでに新パターンで運用されている実際の開始日を、
		// 反映元のstart値とpatternの整合性から求める）。

		"exception": [
			{
				"originalStart": "2026-09-21T10:00:00",
				"originalTimezone": "Asia/Tokyo",
				// このexceptionが本来何の発生を置き換えるものかを示す、変更前の日時
				// （Google: originalStartTime、Graphは相当する専用フィールドがないため
				//  取得時に自前でoccurrence一覧と突き合わせて判定する）

				"status": "updated", // "updated"（内容変更） or "deleted"（キャンセル・削除）

				"id": "def456...", // このexceptionインスタンス自身のプラットフォーム上のID

				"sync": null,
				// 親と同様の同期マーカー。exceptionごとに個別に持つ
				// （親のsyncとは独立。例外は例外同士でのみ突き合わせる）

				"updatedAt": "2026-09-05T09:00:00Z",
				// このexception自身の最終更新日時。親のupdatedAtとは独立して管理・比較する

				"event": {
					// status="deleted"のときはこのオブジェクトの中身は参照しない（無視してよい）
					// この中身には、更新された内容だけ入れることで、処理を軽量にする。親イベントと内容を比較して、更新されていれば設定し、同一であれば設定しない。
					"isAllDay": true,
					"timeZone": "Asia/Tokyo",
					"start": "2026-09-21T14:00:00",
					"end": "2026-09-21T15:00:00",
					"subject": "毎週月曜10時のMTG（時間変更）"
				}
			}
		]
	},

	"raw": {} // イベントの生データを
}
```
